\section{Chains of CSZ factors}

In Lean a factor is a $T$-invariant sub-$\sigma$-algebra (\texttt{Factor}), and the
$\sigma$-algebra generated by a family is their supremum \texttt{sSup} in the lattice of
measurable spaces.  The chain lemma is proved in three steps: the supremum is again a factor;
every set in it is approximated in measure by sets in some member of the chain; and the
multiple correlation of an approximated set is comparable, step by step, to that of a set in
the approximating factor.

\begin{lemma}[The supremum of a chain of factors is a factor]
  \label{lem:sSup-factor}
  \lean{CantorSzemeredi.MPS.Factor.sSup}
  \leanok
  \uses{def:factor}
  Let $\{\Dbb_\alpha\}_{\alpha \in I}$ be a family of factors and $\Dbb \coloneqq \bigvee_\alpha \Dbb_\alpha$
  the $\sigma$-algebra they generate.  Then $\Dbb \subset \Abb$ and $\Dbb$ is $T$-invariant, so
  $\Dbb$ is a factor.
\end{lemma}

\begin{proof}
  \uses{def:factor}
  $\Dbb \subset \Abb$ because each $\Dbb_\alpha \subset \Abb$ and $\Abb$ is a $\sigma$-algebra.
  For invariance, use the comap: for a measurable space structure $\mathcal M$ on $X$,
  $T^{\ast}\mathcal M \coloneqq \{T^{-1}B : B \in \mathcal M\}$ is again a $\sigma$-algebra
  (\texttt{MeasurableSpace.comap}), and $T^{\ast}$ commutes with suprema
  (\texttt{MeasurableSpace.comap\_iSup}).  Invariance of $\Dbb_\alpha$ says
  $T^{\ast}\Dbb_\alpha = \Dbb_\alpha$ (both inclusions, since $T$ is invertible and the Lean
  definition of invariance is the biconditional).  Hence
  $T^{\ast}\Dbb = \bigvee_\alpha T^{\ast}\Dbb_\alpha = \bigvee_\alpha \Dbb_\alpha = \Dbb$, which is the
  biconditional $B \in \Dbb \iff T^{-1}B \in \Dbb$.
\end{proof}

\begin{lemma}[Approximation from a chain]
  \label{lem:chain-approx}
  \lean{CantorSzemeredi.MPS.Factor.exists_mem_chain_symmDiff_lt}
  \leanok
  \uses{def:factor, lem:sSup-factor}
  Let $\{\Dbb_\alpha\}_{\alpha \in I}$ be a non-empty family of sub-$\sigma$-algebras of $\Abb$
  totally ordered by inclusion, and $\Dbb = \bigvee_\alpha \Dbb_\alpha$.  Then for every
  $E \in \Dbb$ and $\eta > 0$ there are $\alpha \in I$ and $E' \in \Dbb_\alpha$ with
  $m(E \,\triangle\, E') < \eta$.
\end{lemma}

\begin{proof}
  \uses{def:factor}
  Let $\mathcal U \coloneqq \bigcup_\alpha \Dbb_\alpha$.  It is an algebra of sets: it contains
  $\emptyset$ and $X$ (the family is non-empty), is closed under complements, and is closed
  under finite unions and intersections because any two members lie in a common $\Dbb_\alpha$
  (total order).  It generates $\Dbb$.  Let $\mathcal G$ be the class of $E \in \Dbb$ with the
  approximation property: for every $\eta > 0$ there is $E' \in \mathcal U$ with
  $m(E \triangle E') < \eta$.  Then $\mathcal G \supset \mathcal U$ (take $E' = E$), and
  $\mathcal G$ is a $\sigma$-algebra:
  \begin{itemize}
    \item complements: $E^c \triangle E'^c = E \triangle E'$;
    \item countable unions: let $E_j \in \mathcal G$, $E = \bigcup_j E_j$, $\eta > 0$.  By
      continuity from below choose $J$ with $m(E \setminus \bigcup_{j \leq J}E_j) < \eta/2$,
      and for each $j \leq J$ choose $E'_j \in \mathcal U$ with
      $m(E_j \triangle E'_j) < \eta/(2(J+1))$.  Then $E' \coloneqq \bigcup_{j\leq J}E'_j \in \mathcal U$
      and $E \triangle E' \subset (E \setminus \bigcup_{j\leq J}E_j) \cup \bigcup_{j \leq J}(E_j \triangle E'_j)$,
      of measure $< \eta$.
  \end{itemize}
  Hence $\mathcal G = \Dbb$.  (Mathlib: this is
  \texttt{exists\_measure\_symmDiff\_lt\_of\_generateFrom\_isSetSemiring} --- an algebra is a
  set semiring, and $\Dbb$ is generated by $\mathcal U$ --- the same lemma that supplied the
  rectangle approximation in \texttt{HilbertSchmidt.lean}; the approximating set lies in the
  closure of $\mathcal U$ under finite disjoint unions, which is $\mathcal U$ itself.)
\end{proof}

\begin{lemma}[Comparison of multiple correlations, one step at a time]
  \label{lem:chain-comparison}
  \lean{CantorSzemeredi.MPS.multiCorr_ge_half_of_condExp}
  \leanok
  \uses{def:factor, lem:condexp-facts}
  Let $\Dbb_0$ be a factor, $E \in \Abb$, $t \geq 1$, $\kappa \coloneqq 1/(2(t+1))$, and put
  $g \coloneqq \Exp(\one_E\mid\Dbb_0)$ and $H \coloneqq \{g > 1 - \kappa\} \in \Dbb_0$.  Then for
  every $d \in \NNz$,
  \begin{equation}
    \label{eq:chain-comparison}
    m\Bigl( \bigcap_{s=0}^{t} T^{-sd}E \Bigr) \geq \tfrac12\, m\Bigl( \bigcap_{s=0}^{t} T^{-sd}H \Bigr) .
  \end{equation}
\end{lemma}

\begin{proof}
  \leanok
  \uses{lem:condexp-facts}
  Let $H_d \coloneqq \bigcap_{s=0}^{t} T^{-sd}H \in \Dbb_0$ ($\Dbb_0$ is $T$-invariant).  For
  each $s$,
  \[
    m\bigl(H_d \setminus T^{-sd}E\bigr) = \int_{H_d} \bigl(1 - \one_E \circ T^{sd}\bigr)\,dm
      = \int_{H_d} \Exp\bigl(1 - \one_E\circ T^{sd} \mid \Dbb_0\bigr)\,dm
      = \int_{H_d} \bigl(1 - g \circ T^{sd}\bigr)\,dm ,
  \]
  the second equality by the defining property of conditional expectation on the
  $\Dbb_0$-set $H_d$, the third by Lemma~\ref{lem:condexp-facts}(c).  For $x \in H_d$ we have
  $T^{sd}x \in H$, so $g(T^{sd}x) > 1 - \kappa$ and the integrand is $< \kappa$; hence
  $m(H_d \setminus T^{-sd}E) \leq \kappa\, m(H_d)$.  Therefore
  \[
    m\Bigl(\bigcap_{s} T^{-sd}E\Bigr) \geq m\Bigl(H_d \cap \bigcap_{s} T^{-sd}E\Bigr)
      \geq m(H_d) - \sum_{s=0}^{t} m\bigl(H_d \setminus T^{-sd}E\bigr)
      \geq \bigl(1 - (t+1)\kappa\bigr) m(H_d) = \tfrac12 m(H_d) .
  \]
  (In Lean, with $\texttt{multiCorr}$ defined as a measure of an intersection, this is the
  statement, and it uses only \texttt{setIntegral\_condExp}, \texttt{condExp\_comp}, and
  \texttt{measure\_diff\_le\_iff\_le\_add}-type inequalities; no products of indicators appear.)
\end{proof}

\begin{lemma}[Chains]
  \label{lem:chain}
  \lean{CantorSzemeredi.MPS.CSZ_sSup_chain}
  \leanok
  \uses{def:csz, def:factor, lem:sSup-factor, lem:chain-approx, lem:chain-comparison}
  Let $\{\Dbb_\alpha\}_{\alpha \in I}$ be a non-empty family of CSZ factors of $(X,\Abb,m,T)$
  totally ordered by inclusion, and let $\Dbb$ be the $\sigma$-algebra generated by their union.
  Then $\Dbb$ is a CSZ factor.
\end{lemma}

\begin{proof}
  \leanok
  \uses{def:csz, def:factor, lem:sSup-factor, lem:chain-approx, lem:chain-comparison}
  $\Dbb$ is a factor by Lemma~\ref{lem:sSup-factor}.  Fix $\Cantor$, $t \geq 1$, $i \geq 0$ and
  $E \in \Dbb$ with $m(E) > 0$; put $\kappa \coloneqq 1/(2(t+1))$.

  \emph{A set in a single factor.}  By Lemma~\ref{lem:chain-approx} choose $\alpha$ and
  $E' \in \Dbb_\alpha$ with $m(E \triangle E') < \kappa^2 m(E)$.  Put
  $g \coloneqq \Exp(\one_E\mid\Dbb_\alpha)$ and $H \coloneqq \{g > 1-\kappa\} \in \Dbb_\alpha$.
  Then $\norm{\one_E - g}_2 \leq \norm{\one_E - \one_{E'}}_2 = m(E\triangle E')^{1/2}$, because
  $g$ is the orthogonal projection of $\one_E$ onto $L^2(\Dbb_\alpha)$ and $\one_{E'}$ lies in
  that subspace (Mathlib: \texttt{condExpL2} is the orthogonal projection, or directly
  $\norm{u - \Exp(u\mid\Dbb_\alpha)}_2 \leq \norm{u - v}_2$ for $\Dbb_\alpha$-measurable $v$).
  So $\norm{\one_E - g}_2^2 < \kappa^2 m(E)$.  We claim $m(H) > 0$.  Otherwise $g \leq 1 - \kappa$
  a.e., so $1 - g \geq \kappa$ a.e.\ on $E$, and
  $\norm{\one_E - g}_2^2 \geq \int_E (1-g)^2\,dm \geq \kappa^2 m(E)$, a contradiction.

  \emph{Comparison and conclusion.}  By Lemma~\ref{lem:chain-comparison} applied to
  $\Dbb_0 = \Dbb_\alpha$ with $d = b^ik_n$, for every $n$,
  \[
    m\Bigl(\bigcap_{s=0}^{t} T^{-sb^ik_n}E\Bigr) \geq \tfrac12\, m\Bigl(\bigcap_{s=0}^{t} T^{-sb^ik_n}H\Bigr) .
  \]
  Average over $M \leq n < N$ and take $\unifliminf$; the uniform liminf is monotone and
  commutes with multiplication by the positive constant $\tfrac12$
  (\texttt{Filter.liminf} of $\tfrac12 a$ is $\tfrac12$ the liminf of $a$).  The right side is
  positive because $H \in \Dbb_\alpha$, $m(H) > 0$, and $\Dbb_\alpha$ is CSZ at the same
  $\Cantor$, $t$, $i$.  As $\Cantor, t, i, E$ were arbitrary, $\Dbb$ is CSZ.
\end{proof}

\begin{remark}
  \eqref{eq:chain-comparison} is the \emph{only} point at which the CSZ property enters, and it
  holds for every step $d$ separately.  That is why the classical argument applies verbatim to
  the CSZ property with its uniform averages, its quantification over all Cantor sets, and its
  dilations: each of those is a statement about \emph{which steps $d$ are averaged}, and the
  comparison is uniform in $d$.  Lemma~\ref{lem:chain-comparison} is that standalone statement.
\end{remark}

\section{The dichotomy}

\begin{lemma}[Dichotomy]
  \label{lem:dichotomy}
  \lean{CantorSzemeredi.MPS.weaklyMixingExtension_or_compactExtension}
  \notready
  \uses{lem:wiener}
  \uses{def:factor, def:cwm, def:compact-extension}
  Let $(X,\Abb,m,T)$ be an ergodic system satisfying the standing assumption, and let
  $\Dbb \neq \Abb$ be a factor.  Then either $\Abb$ is a weakly mixing extension of $\Dbb$, or
  there is a factor $\Ebb$ with $\Dbb \subsetneq \Ebb \subset \Abb$ which is a compact extension
  of $\Dbb$.
\end{lemma}

\begin{proof}
  Due to Furstenberg; see Moreira's notes, Proposition 8.6.1, for a proof in the present
  language.  The proof is for ergodic systems on compact metric spaces, which is the setting
  here, and uses the disintegration of $m$ over $\Dbb$.
\end{proof}

\begin{remark}[The one hard import]
  Lemma~\ref{lem:dichotomy} is quoted in the paper and is by far the largest single item in the
  formalization that is not written out there.  Two points about our definitions make it apply
  as stated, and both must be preserved in Lean:
  \begin{itemize}
    \item Moreira calls $f$ conditionally weakly mixing if $\abs{\cip{T^n f}{g}} \to 0$ in
      uniform Ces\`aro mean, in $L^2(\Dbb)$, for \emph{every} $g$ in the Hilbert module.  Our
      Definition~\ref{def:cwm} is the case $g=f$ with ordinary Ces\`aro means, hence
      \emph{implied} by his.  So a weakly mixing extension in his sense is one in ours, and the
      dichotomy as he proves it gives the dichotomy as we state it.
    \item Our Definition~\ref{def:compact-extension} of conditional compactness, with orbits
      indexed by $n \in \ZZ$ and approximants in the Hilbert module, is exactly his.
  \end{itemize}
  Recommended strategy: state Lemma~\ref{lem:dichotomy} as an axiom-free \texttt{theorem} with
  \texttt{sorry}, behind a clearly documented interface, and complete the rest of the development
  against it.  Discharging it is its own project.

  \smallskip
  \emph{What is in place towards it.}  The proof begins by taking $f$ in the module with
  $\Exp(f\mid\Dbb)=0$ which is \emph{not} conditionally weakly mixing, and producing an atom of
  the spectral measure $\sigma_G$ of $G = f \otimes f$.  The measure-theoretic half of that step
  is formalized: \texttt{exists\_atom\_of\_not\_tendsto\_cesaro\_zero} turns a non-vanishing
  Ces\`aro mean of $\abs{\widehat{\sigma}(n)} ^2$ into an atom.  Notably this needs only
  $(\sigma\times\sigma)(\Delta) = \int \sigma(\{\alpha\})\,d\sigma$, which is Fubini, and not the
  full identity $\sum_\theta \sigma(\{\theta\}) ^2 = (\sigma\times\sigma)(\Delta)$: an integral of
  a non-negative function is zero as soon as the function vanishes identically, so positivity of
  the diagonal exhibits an atom directly.  Together with \texttt{integral\_tensorSq\_comp}, which identifies $\widehat{\sigma_G}(n)$ with
  $\psi(n)$, and the converse estimate
  $\avg{N}\psi \to 0 \Rightarrow$ conditional weak mixing
  (\texttt{condWeaklyMixing\_of\_tendsto\_cesaro\_psi}, the triangle inequality
  \texttt{eLpNorm\_sum\_le} followed by Cauchy--Schwarz), \textbf{that first step is now
  proved}, as \texttt{exists\_atom\_of\_not\_condWeaklyMixing}.

  \smallskip
  \emph{The Hilbert--Schmidt step.}  Its first half is also formalized:
  \texttt{exists\_eigenvector\_of\_atom} turns an atom of the spectral measure at $\theta$ into a
  genuine eigenvector, $Uw = e(\theta)w$ with $w \neq 0$.  Von Neumann's mean ergodic theorem
  (which Mathlib has) makes the Birkhoff averages of the contraction $e(-\theta)U$ converge to
  the orthogonal projection onto its fixed points, and
  \texttt{tendsto\_norm\_birkhoff\_sq} --- the quadratic form identity \texttt{norm\_sum\_sq}
  followed by dominated convergence against the kernel $\abs{\mathrm{ces}_N(\alpha-\theta)} ^2$
  --- shows the limit has squared norm $\sigma(\{\theta\}) \neq 0$.

  The missing theory has since been supplied.  Mathlib has no Hilbert--Schmidt operators ---
  only Gram--Schmidt orthogonalization, which is unrelated --- so
  \texttt{CantorSzemeredi/Analysis/HilbertSchmidt.lean} develops them from scratch, written so
  that it can be upstreamed: the operator $T_k f(x) = \int k(x,y) f(y)\,d\nu(y)$ attached to a
  kernel $k \in L^2(\mu\otimes\nu)$, the bound $\norm{T_k f}_2 \le \norm{k}_2 \norm{f}_2$
  (\texttt{eLpNorm\_integralKernel\_le}), its packaging as a continuous linear map
  (\texttt{integralKernelL}), and the theorem that these operators are compact
  (\texttt{isCompactOperator\_integralKernelL}).  The compactness proof has no analogue in
  Mathlib either: a tensor product $g \otimes h$ gives a rank-one operator, and rank-one
  operators are compact (\texttt{isCompactOperator\_rankOne}, also absent upstream, which has
  the closure properties but no base case); the tensor products are total in
  $L^2(\mu\otimes\nu)$ because a function orthogonal to every one of them integrates to zero
  over every measurable rectangle, hence over every measurable set by Dynkin's lemma; and
  $k \mapsto T_k$ is continuous by the Hilbert--Schmidt bound, so compactness passes to the
  closure.

  \smallskip
  \emph{The fibrewise reading.}  This too is now formalized, in
  \texttt{CantorSzemeredi/Dynamics/Fibre.lean}.  Since
  $\lambda = \int m_x \otimes m_x \,dm(x)$ is an \emph{average} of product measures, Fubini for
  the joining (\texttt{lintegral\_joining}) makes the fibre integrals
  $\int \abs{H}^2 \,d(m_x\otimes m_x)$ average to $\norm{H}_{L^2(\lambda)}^2 < \infty$, so they
  are finite for $m$-a.e.\ $x$: the eigenfunction is, on almost every fibre, a Hilbert--Schmidt
  kernel (\texttt{ae\_memLp\_fibre}).  Each $m_x$ is a probability measure, so
  \texttt{isCompactOperator\_integralKernelL} applies fibre by fibre and gives
  \texttt{ae\_isCompactOperator\_fibre}: for a.e.\ $x$, $H$ defines a compact operator on
  $L^2(m_x)$.

  One point of care, recorded in that file: $H \in L^2(\lambda)$ gives only
  $\lambda$-a.e.\ strong measurability, which says nothing about any individual fibre --- a
  $\lambda$-null set can meet every fibre.  The lemmas therefore take a strongly measurable
  representative, which costs nothing, since \texttt{integralKernelL\_congr} says the operator
  depends only on the a.e.\ class of its kernel.

  \smallskip
  \emph{From fibre operators to the module.}  The fibre operators assemble into a single
  operator on the Hilbert module,
  $(\Phi_H a)(x) = \int H(x,y) a(y) \,dm_x(y)$, and
  \texttt{CantorSzemeredi/Dynamics/CondHS.lean} proves the estimate that governs it.  Two
  ingredients: \texttt{condKernel\_comp\_measure}, the disintegration $m = \int m_x \,dm(x)$
  (Mathlib proves this against the \emph{trimmed} measure, and since the kernel is
  $\Dbb$-measurable, \texttt{lintegral\_trim} removes the trim), whose consequence
  \texttt{ae\_ae\_condKernel} is what lets an a.e.\ statement about the base be used on almost
  every fibre; and \texttt{abs\_condIntegralOp\_le}, Cauchy--Schwarz on a single fibre.
  Together they give
  \[
    \norm{\Phi_H a}_{\Ebb|\Dbb} \;\leq\; C \, \norm{H}_{\mathrm{HS}, \text{fibre}}
    \qquad\text{whenever } \norm{a}_{\Ebb|\Dbb} \leq C
  \]
  (\texttt{condNorm\_condIntegralOp\_le}), and hence
  \texttt{condCompact\_of\_kernel\_net}: if the orbit of $f$ is realised as $\Phi_{K_n} a$ for a
  family of kernels admitting finite nets in the fibre Hilbert--Schmidt norm, then $f$ is
  conditionally compact in the sense of Definition~\ref{def:compact-extension}.  Applied to a
  \emph{difference} of kernels, the estimate is exactly what converts an approximation of
  kernels into an approximation of functions --- which is the shape \texttt{CondCompact} asks
  for, since its finite net must consist of module elements and must serve every $n \in \ZZ$
  at once.

  Note that these estimates carry their integrability hypotheses explicitly, collected in
  \texttt{IsFibreHS}.  That is not bureaucracy: $H \in L^2(\lambda)$ gives only a.e.\
  statements about the base, and a $\lambda$-null set can meet every fibre, so nothing
  fibrewise follows until it is pushed through \texttt{ae\_ae\_condKernel}.

  \smallskip
  \emph{The finite net.}  \texttt{condCompact\_of\_kernel\_net} is a reduction: it assumes a
  finite net of kernels.  Producing that net needs no measurable selection, contrary to what
  one might expect from the fibrewise compactness --- the orbit is not a general precompact
  set but a \emph{circle}.  Since $H$ is an eigenfunction, $H \circ R^n = e(n\theta) H$, so
  the orbit of kernels traces $(\cos n\theta, \sin n\theta)$ through the real span of
  $\Re H$ and $\Im H$.  The net is then total boundedness of a bounded subset of $\RR^2$,
  a fact about $\RR^2$ and not about the fibres, carried over by the two estimates
  \[
    \norm{c H}_{\mathrm{HS},\text{fibre}} = \abs{c}\,\norm{H}_{\mathrm{HS},\text{fibre}},
    \qquad
    \norm{H+K}_{\mathrm{HS},\text{fibre}}
      \leq \norm{H}_{\mathrm{HS},\text{fibre}} + \norm{K}_{\mathrm{HS},\text{fibre}} .
  \]
  The second is Minkowski applied twice --- inside each fibre for the rows, then across the
  fibre for the row norms (\texttt{hsFibre\_add\_le}); both uses go through
  \texttt{CantorSzemeredi/Analysis/SqrtIntegral.lean}, which states Cauchy--Schwarz and
  Minkowski for $\sqrt{\int f^2}$ rather than for \texttt{eLpNorm}, since a conditional norm
  is a real number depending on a base point which is then integrated again.  The outcome is
  \texttt{exists\_kernel\_net}, and combining it with the reduction gives
  \texttt{condCompact\_of\_param\_orbit}: an orbit of kernels drawn from a bounded
  two-parameter family, acting on a single module element, consists of conditionally compact
  functions.

  \smallskip
  \emph{The regularity is not an extra assumption.}  The measurability obligations follow
  from joint measurability of $H$: both $x \mapsto \norm{H(x,\cdot)}_{L^2(m_x)}$ and
  $\Phi_H a$ are Bochner integrals against the kernel, and Mathlib's
  \texttt{StronglyMeasurable.integral\_kernel\_prod\_right'} makes those measurable in the
  base point (\texttt{measurable\_hsRow}, \texttt{measurable\_condIntegralOp}).  The
  remaining per-fibre conditions reduce to conditions on the base: by the disintegration, a
  function in $L^2(m)$ is in $L^2(m_x)$ for a.e.\ $x$ (\texttt{ae\_memLp\_condKernel}).  So
  \texttt{IsFibreReg} and \texttt{IsFibreHS} are built by
  \texttt{isFibreReg\_of\_memLp\_hsRow} and \texttt{isFibreHS\_of\_memLp} from joint
  measurability, a condition on the rows, and global $L^2(m)$ bounds --- nothing fibrewise
  has to be checked by hand.

  \smallskip
  \emph{Constancy along fibres.}  The row condition --- that $\int H(z,y)^2\,dm_z(y)$ is
  finite for a.e.\ $z$ --- does \emph{not} follow from \texttt{ae\_memLp\_fibre}, which
  controls $\int\!\!\int H^2\,d(m_x \otimes m_x)$, an integral over a fibre \emph{pair}.
  Relating the two is the a.e.\ constancy of $z \mapsto m_z$ along fibres, and that is now
  proved (\texttt{ae\_ae\_integral\_condKernel\_eq}).  The argument needs neither countable
  generation nor measurable selection.  Fix a bounded measurable $g$ and put
  $G(x) = \int g\,dm_x$.  Since the kernel is $\Dbb$-measurable, so is $G$, hence $G$ is its
  own conditional expectation and $\int G\,dm_x = G(x)$ a.e.; the same applies to $G^2$,
  because that is $\Dbb$-measurable too.  Subtracting, the variance of $G$ over the fibre
  through $x$ vanishes, so $G$ is constant there.

  Applied one indicator at a time, this identifies two measures on $X \times X$: the graph
  measure $\rho = \int \delta_z \otimes m_z \,dm(z)$ equals the joining $\lambda$
  (\texttt{compProd\_condKernel\_eq\_joining}) --- drawing $z$ and then $y$ from $m_z$ has
  the same law as drawing a fibre and then both coordinates independently from it.  Only the
  functional form of constancy is used, one bounded $g = \one_B$ at a time, which is why no
  countable generating $\pi$-system is needed.  The row condition
  (\texttt{ae\_memLp\_row}) and the bound on the row norms (\texttt{memLp\_hsRow}) both drop
  out, so \texttt{isFibreReg\_of\_memLp\_joining} builds the whole of \texttt{IsFibreReg}
  from nothing but joint measurability of $H$ and $H \in L^2(\lambda)$.

  \smallskip
  \emph{The orbit identity.}  The conditional operator intertwines the dynamics:
  \[
    (\Phi_H a) \circ T^n \;=\; \Phi_{H \circ R^n}\,(a \circ T^n), \qquad R = T \times T
  \]
  (\texttt{condIntegralOp\_comp\_T} and \texttt{condIntegralOp\_comp\_iterate}).  The proof is
  short given what precedes it: both sides integrate row by row against the conditional
  measures, so by \texttt{compProd\_condKernel\_eq\_joining} both are integrals against
  $\lambda$, and the identity becomes the $R$-invariance of $\lambda$.  In particular no
  equivariance statement relating $m_{Tx}$ to $m_x$ is needed --- which matters, because the
  set-level \texttt{condKernel\_comp} does not give one without countable generation.

  \smallskip
  \emph{Reduction to finite-rank kernels.}  The orbit identity makes one thing explicit that
  was not visible before: along the orbit the transformation acts on the kernel \emph{and} on
  the function.  The kernel part is harmless --- for an eigenfunction $H \circ R^n$ runs
  around a circle, which \texttt{exists\_kernel\_net} handles --- but $a \circ T^n$ moves too,
  so \texttt{condCompact\_of\_param\_orbit}, which assumes a \emph{fixed} $a$, does not apply
  to an eigenfunction orbit; its statement now says so.

  What is needed instead is that $\Phi_H$ carries the bounded set $\{a \circ T^n\}$ to a
  conditionally totally bounded set, and \texttt{condCompact\_of\_kernel\_approx} reduces this
  to the finite-rank case: the conditional Hilbert--Schmidt bound transports an
  essential-sup approximation of \emph{kernels} to the \emph{functions}, the triangle
  inequality \texttt{condNorm\_add\_le} adds the two errors, and a finite-rank kernel acts by
  $\Phi_{\sum g_i \otimes h_i} b = \sum_i \cip{h_i}{b} \, g_i$
  (\texttt{condIntegralOp\_sum\_tensor}), with $\Dbb$-measurable coefficients.

  \smallskip
  \emph{What is still open, and a question about the source.}  The reduction leaves two
  hypotheses, and neither is a formality.

  \emph{(i)} It asks for $\norm{H-K}_{\mathrm{HS},\text{fibre}} \leq \delta$ \emph{a.e.\ on
  the base} with $K$ of finite rank, whereas $L^2(\lambda)$ density of tensor products gives
  smallness of the $L^2$ norm \emph{over} the base.

  \emph{(ii)} It asks for a finite net covering the finite-rank images $\sum_i c_i^n g_i$,
  where $c_i^n = \cip{h_i}{a \circ T^n}$ are bounded $\Dbb$-measurable \emph{functions}.
  Under the \emph{old} form of Definition~\ref{def:compact-extension} this meant a finite net
  for $\{c^n\}$ in the $L^{\infty}(\Dbb)$ ball, which does not exist.  With the definition
  corrected to Moreira's it dissolves: the index may depend on the fibre, and rounding
  $c^n(x)$ to a fixed finite grid pointwise is exactly such a choice, the finitely many grid
  points giving the finitely many approximants $\sum_i v_i g_i$.  See
  Remark~\ref{rem:compact-mismatch}.

  \smallskip
  \emph{(i) is settled by a cutoff.}  The bridge is the exact identity
  \[
    \int_X \norm{H}_{\mathrm{HS},\text{fibre}}^2 \,dm \;=\; \norm{H}_{L^2(\lambda)}^2
  \]
  (\texttt{lintegral\_hsFibre\_sq}), which is the disintegration applied twice --- once to
  unfold the fibre, once to recognise the row integrals as an integral against $\lambda$.  So
  $L^2(\lambda)$ approximation \emph{does} control $\norm{H-K}_{\mathrm{HS},\text{fibre}}$,
  but only in $L^2$ of the base; Chebyshev converts this to smallness outside a set of small
  measure.  The point is that the set may be taken in $\Dbb$, since
  $\norm{\cdot}_{\mathrm{HS},\text{fibre}}$ is $\Dbb$-measurable
  (\texttt{stronglyMeasurable\_hsFibre}) --- the kernel $x \mapsto m_x$ being
  $\Dbb$-measurable --- and a $\Dbb$-set is a union of fibres, so cutting the rows down to it
  simply multiplies the fibre norm by its indicator
  (\texttt{ae\_ae\_indicator\_eq}, \texttt{hsFibre\_indicator}).  The cut kernel then
  satisfies the bound \emph{everywhere}: this is \texttt{exists\_cutoff\_hsFibre\_le}, with
  $m(B^c) \leq \norm{H-K}_{L^2(\lambda)}^2/\delta^2$.  The discarded set is where the
  density in \emph{compact extension} absorbs the error, exactly as anticipated.

  \smallskip
  \emph{The density.}  The input to the cutoff --- that finite-rank kernels are dense in
  $L^2(\lambda)$ --- is now in place.  The repository already had
  \texttt{topologicalClosure\_span\_prodKernels}, but for complex kernels and a \emph{product}
  measure, and $\lambda$ is not a product.  The Dynkin argument behind it never used the
  product structure, only that measurable rectangles form a $\pi$-system generating the
  product $\sigma$-algebra, so
  \texttt{ae\_eq\_zero\_of\_forall\_setIntegral\_rectangle\_eq\_zero} is now stated for an
  arbitrary finite measure on $X \times X$ and any Banach target.  What does \emph{not}
  generalise is \texttt{memLp\_prod\_mul}: off a product measure
  $\norm{g \otimes h}_{L^2}$ is not controlled by $\norm{g}_2\norm{h}_2$, so the totality is
  proved for \emph{indicators} of rectangles --- bounded, hence in $L^2$ of any finite
  measure --- which is all that is needed
  (\texttt{topologicalClosure\_span\_rectKernels}).  \texttt{exists\_rectSum\_eLpNorm\_sub\_lt}
  extracts from this an explicit finite sum $\sum_i c_i \one_{A_i} \otimes \one_{B_i}$.

  Combining with the cutoff gives \texttt{exists\_finiteRank\_cutoff}: for every
  $\delta, \eta > 0$ a finite-rank kernel $K$ and a $\Dbb$-set $B$ with $m(B^c) \leq \eta$ on
  which $\norm{H-K}_{\mathrm{HS},\text{fibre}} \leq \delta$ everywhere.  This is exactly the
  hypothesis of \texttt{condCompact\_of\_kernel\_approx}, with the exceptional set $B^c$ left
  for the density in \emph{compact extension} to absorb.

  \smallskip
  \emph{The finite net, and the moving kernel.}  Obstruction (ii) is
  \texttt{exists\_net\_of\_finite\_span}: given finitely many generators $u_i$ of conditional
  norm at most $1$, the $\Dbb$-linear combinations $\sum_i \gamma_i u_i$ with
  $\abs{\gamma_i} \leq M$ are covered, to within $\epsilon$, by the finitely many
  combinations with \emph{constant} coefficients from a grid --- the approximant found by
  rounding $\gamma(x)$ pointwise, so its index depends on the fibre, which is what Moreira's
  definition permits.  The ingredients are the pull-out property
  (\texttt{condNorm\_mul\_left}), subadditivity over a finite sum
  (\texttt{condNorm\_sum\_le}), and \texttt{condNorm\_indicator\_le\_one}, which makes
  indicators admissible generators; \texttt{condIntegralOp\_sum\_tensor} puts a finite-rank
  image in exactly that form, with coefficients $c_i \cip{\one_{B_i}}{b}$.

  The moving kernel turns out not to need separate treatment.
  \texttt{condCompact\_of\_kernel\_approx} is now stated for a kernel \emph{family} $H_n$ ---
  nothing in its proof required the kernel fixed, since the Hilbert--Schmidt bound is applied
  at each $n$ --- which is the shape \texttt{condIntegralOp\_comp\_iterate} produces.  And for
  an eigenfunction $H \circ R^n = \mathrm{ker}(u_n)$ with $u_n$ on the unit circle, so
  \texttt{condIntegralOp\_kerComb} splits $\Phi_{K_n}(b_n)$ into
  $u_n^{(1)}\Phi_{K_0}(b_n) + u_n^{(2)}\Phi_{K_1}(b_n)$; expanding each by
  \texttt{condIntegralOp\_sum\_tensor} gives a combination of the \emph{fixed} generators
  $\one_{A_i}$ with coefficients $u_n^{(j)} c_i \cip{\one_{B_i}}{b_n}$, still
  $\Dbb$-measurable and still bounded.  So the circle is absorbed into the coefficients that
  the grid rounds: one net covers the whole orbit, and no two-term net is required.
  \texttt{kerComb\_sub\_kerComb} and \texttt{hsFibre\_kerComb\_le} supply the matching bound
  on $H_n - K_n$.

  \smallskip
  \smallskip
  \emph{Both hypotheses are now theorems.}  \texttt{exists\_finiteRank\_cutoff} supplies the
  kernel approximation, and \texttt{exists\_net\_of\_finiteRank} the net:
  \texttt{condIntegralOp\_rectSum} identifies the image of a rectangle-sum kernel as a
  $\Dbb$-linear combination of the fixed generators $\one_{A_i}$ with coefficients
  $c_i \cip{\one_{B_i}}{b}$ --- $\Dbb$-measurable by construction, bounded by
  conditional Cauchy--Schwarz against \texttt{condNorm\_indicator\_le\_one} --- and
  \texttt{exists\_net\_of\_finite\_span} rounds those coefficients.

  \smallskip
  \emph{What is still open, and one wrinkle.}  Combining the two is the remaining step, and
  it is not quite mechanical.  \texttt{exists\_finiteRank\_cutoff} produces a $\Dbb$-set $B$
  and the kernel cut to it; since $B \in \Dbb$ the indicator pulls out of the fibre integral,
  so the cut operator is $\one_B \Phi_H a$.  But conditional compactness of that function
  concerns \emph{its} orbit,
  $(\one_B \Phi_H a) \circ T^n = \one_{T^{-n}B}\,\Phi_{H \circ R^n}(a \circ T^n)$, and
  $T^{-n}B$ moves: one cannot cut once and forget it.  What saves the argument is that
  $\Dbb$ is $T$-invariant, so each $\one_{T^{-n}B}$ is again $\Dbb$-measurable and bounded by
  $1$; the coefficients become
  $\one_{T^{-n}B}(x)\, c_i \cip{\one_{B_i}}{b_n}(x)$, and the grid-rounding absorbs
  the moving indicator exactly as it absorbs the circle.  The care needed is in the
  kernel-error term, where the bound must be taken for the cut kernel at each $n$.

  After that, absolute continuity of the integral makes
  $\norm{\Phi_H a - \one_B \Phi_H a}_2$ small, which the density in \emph{compact extension}
  absorbs, and the factor $\Ebb$ can be assembled from the conditionally compact
  functions.
\end{remark}

\section{The structure theorem for ergodic systems}

\begin{theorem}
  \label{thm:ecsz}
  \lean{CantorSzemeredi.MPS.CSZ_of_ergodic}
  \uses{lem:chain, lem:dichotomy, lem:extensions, lem:trivial-csz, def:csz}
  Every ergodic system satisfying the standing assumption is CSZ.
\end{theorem}

\begin{proof}
  \uses{lem:chain, lem:dichotomy, lem:extensions, lem:trivial-csz}
  Let $\mathbf K$ be the set of CSZ factors of $(X,\Abb,m,T)$, partially ordered by inclusion
  of $\sigma$-algebras ($\Dbb \leq \Dbb'$ iff $\Dbb \subset \Dbb'$ as sets of subsets of $X$;
  in Lean, \texttt{D.\ensuremath{\sigma} \ensuremath{\leq} D'.\ensuremath{\sigma}}).

  \emph{Zorn.}  $\mathbf K$ is non-empty: the trivial factor is CSZ (Lemma~\ref{lem:trivial-csz}).
  Every chain $\mathbf C \subset \mathbf K$ has an upper bound in $\mathbf K$: if $\mathbf C$
  is empty, the trivial factor; otherwise the supremum $\bigvee \mathbf C$, which is a CSZ factor
  by Lemma~\ref{lem:chain} and contains every member of $\mathbf C$.  By Zorn's lemma
  (\texttt{zorn\_le\_nonempty} or \texttt{zorn\_le}), $\mathbf K$ has a maximal element $\Dbb$:
  a CSZ factor not strictly contained in any CSZ factor.

  \emph{Maximality forces $\Dbb = \Abb$.}  Suppose $\Dbb \neq \Abb$, i.e.\ $\Dbb$ is not the
  whole $\sigma$-algebra.  Apply Lemma~\ref{lem:dichotomy}.
  \begin{itemize}
    \item If $\Abb$ is a weakly mixing extension of $\Dbb$, then $\Abb$ is CSZ by
      Lemma~\ref{lem:extensions}(W), i.e.\ the top factor is in $\mathbf K$; it strictly
      contains $\Dbb$, contradicting maximality.
    \item Otherwise there is a factor $\Ebb$ with $\Dbb \subsetneq \Ebb \subset \Abb$ which is a
      compact extension of $\Dbb$; it is CSZ by Lemma~\ref{lem:extensions}(K), and strictly
      contains $\Dbb$, contradicting maximality.
  \end{itemize}
  Hence $\Dbb = \Abb$, which says that $(X,\Abb,m,T)$ --- the top factor --- is CSZ.
\end{proof}

\begin{remark}[Factors up to null sets]
  Definition~\ref{def:factor} identifies factors agreeing up to null sets, but the Lean
  \texttt{Factor} is a literal $\sigma$-algebra and the order above is literal inclusion.  No
  harm results.  The one place where the distinction could matter is the hypothesis
  $\Dbb \neq \Abb$ of the dichotomy, taken literally.  If $\Dbb \subsetneq \Abb$ literally but
  every $\Abb$-set is a.e.\ equal to a $\Dbb$-set, then every $f \in L^2(\Abb)$ with
  $\Exp(f\mid\Dbb) = 0$ is a.e.\ zero (it is a.e.\ equal to a $\Dbb$-measurable function, which
  is then its own conditional expectation), so $\Abb$ is vacuously a weakly mixing extension of
  $\Dbb$, and the first branch applies.  So the literal statement of Lemma~\ref{lem:dichotomy}
  is consistent, and no quotient by null sets is needed anywhere in the Zorn argument.
\end{remark}

\section{Reduction to ergodic systems on compact metric spaces}

The standing assumption (compact metric $X$, Borel $\Abb$) is used in two ways: through the
disintegration of $m$ over a factor, which needs $X$ standard Borel
(\texttt{StandardBorelSpace} in Lean), and through the ergodicity hypothesis of the
dichotomy.  The reduction removes both.  It has three ingredients: a symbolic model, the
ergodic decomposition, and a Fatou argument adapted to uniform liminfs.

\begin{lemma}[Symbolic model]
  \label{lem:symbolic-model}
  \lean{CantorSzemeredi.MPS.symbolicModel, CantorSzemeredi.MPS.multiCorr_symbolicModel}
  \uses{def:mps, def:csz}
  Let $(X,\Abb,m,T)$ be an invertible measure preserving system and $A \in \Abb$.  Let
  $\Omega \coloneqq \{0,1\}^{\ZZ}$ with the product topology and its Borel $\sigma$-algebra
  $\mathcal B$, and let $\tau : \Omega \to \Omega$, $(\tau\omega)(n) = \omega(n+1)$.  Define
  $\Phi : X \to \Omega$ by $\Phi(x)(n) \coloneqq \one_A(T^nx)$, $n \in \ZZ$, and
  $\widetilde A \coloneqq \{\omega : \omega(0) = 1\}$.  Then:
  \begin{enumerate}
    \item[(i)] $\Omega$ is a compact metrizable space, $\tau$ is a homeomorphism, and
      $\mathcal B$ is the product $\sigma$-algebra; in particular $(\Omega,\mathcal B)$ is
      standard Borel and $\widetilde A$ is clopen.
    \item[(ii)] $\Phi$ is measurable and $\Phi \circ T = \tau \circ \Phi$.
    \item[(iii)] $\mu \coloneqq \Phi_\ast m$ is a $\tau$-invariant Borel probability measure,
      so $(\Omega,\mathcal B,\mu,\tau)$ is an invertible measure preserving system.
    \item[(iv)] $\Phi^{-1}(\tau^{-d}\widetilde A) = T^{-d}A$ for every $d \in \ZZ$, hence for all
      $d_0,\dots,d_t \in \ZZ$,
      \begin{equation}
        \label{eq:model}
        m\Bigl( \bigcap_{s=0}^{t} T^{-d_s}A \Bigr) = \mu\Bigl( \bigcap_{s=0}^{t} \tau^{-d_s}\widetilde A \Bigr) ,
      \end{equation}
      and $\mu(\widetilde A) = m(A)$.
  \end{enumerate}
\end{lemma}

\begin{proof}
  \uses{def:mps}
  (i) $\{0,1\}$ is finite discrete, so a countable product is compact (Tychonoff;
  \texttt{Pi.compactSpace}) and metrizable (countable product of metrizable spaces;
  \texttt{Pi} instances in Mathlib's \texttt{Topology.Metrizable}), hence Polish and standard
  Borel.  The shift is continuous with continuous inverse $(\tau^{-1}\omega)(n) = \omega(n-1)$.
  The Borel $\sigma$-algebra of a countable product of second countable spaces is the product
  $\sigma$-algebra (\texttt{Pi.borelSpace}).  $\widetilde A$ is the preimage of the open and
  closed set $\{1\}$ under the continuous coordinate projection.

  (ii) Each coordinate $x \mapsto \one_A(T^nx)$ is measurable, so $\Phi$ is measurable into the
  product $\sigma$-algebra (\texttt{measurable\_pi\_iff}).  And
  $\Phi(Tx)(n) = \one_A(T^{n+1}x) = \Phi(x)(n+1) = (\tau\Phi(x))(n)$.

  (iii) $\mu$ is a probability measure as the pushforward of one.  For $B \in \mathcal B$,
  $\mu(\tau^{-1}B) = m(\Phi^{-1}\tau^{-1}B) = m((\tau\circ\Phi)^{-1}B) = m((\Phi\circ T)^{-1}B)
   = m(T^{-1}\Phi^{-1}B) = m(\Phi^{-1}B) = \mu(B)$, using (ii) and invariance of $m$.  In Lean:
  \texttt{MeasurePreserving} of $\tau$ with respect to \texttt{Measure.map \ensuremath{\Phi} m}
  follows from \texttt{MeasurePreserving.comp} and \texttt{Measure.map\_map}.

  (iv) $x \in \Phi^{-1}(\tau^{-d}\widetilde A)$ iff $(\tau^d\Phi(x))(0) = 1$ iff
  $\Phi(x)(d) = 1$ iff $T^dx \in A$ iff $x \in T^{-d}A$.  Preimages commute with
  intersections, so \eqref{eq:model} follows from the definition of $\mu$; $d_0 = 0$ gives
  $\mu(\widetilde A) = m(A)$.
\end{proof}

\begin{lemma}[Ergodic decomposition]
  \label{lem:ergodic-decomposition}
  \lean{CantorSzemeredi.MPS.ergodicDecomposition}
  \uses{def:mps, def:factor, def:joining}
  Let $(\Omega,\mathcal B,\mu,\tau)$ be an invertible measure preserving system on a standard
  Borel space with countably generated $\mathcal B$.  There is a measurable family
  $(\mu_y)_{y \in \Omega}$ of Borel probability measures (a Markov kernel $y \mapsto \mu_y$)
  such that:
  \begin{enumerate}
    \item[(i)] $\mu(B) = \int_\Omega \mu_y(B)\,d\mu(y)$ for every $B \in \mathcal B$; more
      generally $\int u\,d\mu = \int\!\!\int u\,d\mu_y\,d\mu(y)$ for bounded measurable $u$.
    \item[(ii)] For $\mu$-a.e.\ $y$, $\mu_y$ is $\tau$-invariant.
    \item[(iii)] For $\mu$-a.e.\ $y$, $\mu_y$ is ergodic: every $\tau$-invariant $B \in \mathcal B$
      has $\mu_y(B) \in \{0,1\}$.
  \end{enumerate}
\end{lemma}

\begin{proof}
  \uses{def:factor, def:joining}
  Mathlib has no ergodic decomposition; here is a route built from pieces that are either in
  Mathlib or already in this project.

  \emph{The kernel.}  Let $\mathcal I \subset \mathcal B$ be the $\sigma$-algebra of
  $\tau$-invariant sets, $\mathcal I = \{B : \tau^{-1}B = B\}$; it is a factor.  Since
  $\Omega$ is standard Borel, Mathlib's \texttt{ProbabilityTheory.condExpKernel \ensuremath{\mu} \ensuremath{\mathcal I}}
  provides a kernel $y \mapsto \mu_y$ with $\int \one_B\,d\mu_y = \Exp(\one_B\mid\mathcal I)(y)$
  for $\mu$-a.e.\ $y$, for each $B$; and $\mu = \int \mu_y\,d\mu(y)$ is
  \texttt{condKernel\_comp\_measure} of this project (the disintegration, with the trim
  removed), which is (i).

  \emph{Invariance of a.e.\ fibre.}  Let $\mathcal P$ be a countable $\pi$-system generating
  $\mathcal B$ (for $\Omega = \{0,1\}^{\ZZ}$, the cylinder sets).  For $B \in \mathcal P$ the
  functions $\one_{\tau^{-1}B} = \one_B \circ \tau$ and $\one_B$ have the same conditional
  expectation on $\mathcal I$: $\Exp(\one_B\circ\tau\mid\mathcal I) = \Exp(\one_B\mid\mathcal I)\circ\tau
   = \Exp(\one_B\mid\mathcal I)$, by Lemma~\ref{lem:condexp-commutes} and because an
  $\mathcal I$-measurable function is $\tau$-invariant.  Hence $\mu_y(\tau^{-1}B) = \mu_y(B)$
  for a.e.\ $y$, for each of the countably many $B \in \mathcal P$, so for a.e.\ $y$ for all of
  them at once.  For such $y$ the two probability measures $\mu_y \circ \tau^{-1}$ and $\mu_y$
  agree on the generating $\pi$-system $\mathcal P$ and so agree
  (\texttt{MeasureTheory.ext\_of\_generate\_finite}).  This is (ii).

  \emph{Ergodicity of a.e.\ fibre.}  Let $P_\nu$ denote, for a $\tau$-invariant probability
  measure $\nu$, the orthogonal projection of $L^2(\nu)$ onto the closed subspace of
  $\tau$-invariant functions.  By the mean ergodic theorem
  (\texttt{ContinuousLinearMap.tendsto\_birkhoffAverage\_orthogonalProjection}),
  $\avg{n<N} u\circ\tau^n \to P_\nu u$ in $L^2(\nu)$ for every $u \in L^2(\nu)$.
  Three observations.
  \begin{enumerate}
    \item[(a)] \emph{$\nu$ is ergodic iff $P_\nu u$ is a.e.\ constant for every $u$ in a set
      total in $L^2(\nu)$.}  If $B$ is invariant then $\one_B = P_\nu\one_B$; if the $P_\nu u$ are
      constant for $u$ in a total set then by continuity and linearity of $P_\nu$ its range
      consists of constants, so $\one_B$ is a.e.\ constant and $\nu(B) \in \{0,1\}$.  The
      indicators of the sets in the algebra generated by $\mathcal P$ are total in $L^2(\nu)$
      for every $\nu$ (simple functions over a generating algebra are dense, by
      Lemma~\ref{lem:chain-approx} applied to the constant chain plus density of simple
      functions).
    \item[(b)] \emph{For $u$ bounded, $P_\mu u = \Exp(u\mid\mathcal I)$.}  Both are the
      orthogonal projection onto $L^2(\mathcal I)$, which is the space of invariant functions.
    \item[(c)] \emph{For a.e.\ $y$, $P_{\mu_y}u = \Exp(u\mid\mathcal I)$ in $L^2(\mu_y)$, for each
      bounded $u$.}  By (i) and the mean ergodic theorem for $\mu$,
      $\int\!\!\int \abs{\avg{n<N}u\circ\tau^n - \Exp(u\mid\mathcal I)}^2\,d\mu_y\,d\mu(y) \to 0$,
      so along a subsequence $N_k$ the inner integral tends to $0$ for a.e.\ $y$.  For such $y$,
      the sequence $\avg{n<N_k}u\circ\tau^n$ converges in $L^2(\mu_y)$ both to
      $\Exp(u\mid\mathcal I)$ and (by the mean ergodic theorem for $\mu_y$, which is invariant by
      (ii)) to $P_{\mu_y}u$; limits agree.
  \end{enumerate}
  Finally, $\Exp(u\mid\mathcal I)$ is $\mathcal I$-measurable, and this project's constancy
  lemma (\texttt{ae\_ae\_integral\_condKernel\_eq}, the variance argument) shows that an
  $\mathcal I$-measurable bounded function is $\mu_y$-a.e.\ constant for $\mu$-a.e.\ $y$.
  Apply (c) and this to the countably many $u = \one_B$, $B$ in the algebra generated by
  $\mathcal P$; off a null set of $y$, all $P_{\mu_y}\one_B$ are constant, and (a) gives (iii).
\end{proof}

\begin{remark}[Can ergodicity be avoided?]
  The ergodicity hypothesis of Lemma~\ref{lem:dichotomy} is inherited from the source; the
  proof sketched in the remark following that lemma --- an atom of $\sigma_{f\otimes f}$, an
  eigenfunction on the joining, the Hilbert--Schmidt operator, the conditionally compact
  functions --- does not visibly use it.  If the formal proof of the dichotomy ends up
  ergodicity-free, Lemma~\ref{lem:ergodic-decomposition} and the Fatou step below become
  unnecessary and Proposition~\ref{prop:reduction} reduces to Lemma~\ref{lem:symbolic-model}
  alone, which supplies the standard Borel hypothesis.  Until then, the reduction is as
  written.
\end{remark}

\begin{lemma}[Fatou for uniform liminfs]
  \label{lem:fatou-uniform}
  \lean{CantorSzemeredi.uniformLiminf_integral_ge}
  \uses{def:uniform-liminf}
  Let $(Y,\nu)$ be a probability space and $c : \NNz \times Y \to [0,1]$ with
  $y \mapsto c_n(y)$ measurable for each $n$.  Then
  \[
    \unifliminf \unifavg \int_Y c_n(y)\,d\nu(y) \;\geq\; \int_Y \Bigl( \unifliminf \unifavg c_n(y) \Bigr) d\nu(y) .
  \]
\end{lemma}

\begin{proof}
  \uses{def:uniform-liminf}
  Write $c \coloneqq \unifliminf \unifavg \int c_n\,d\nu$ and $c(y)$ for the integrand on
  the right; note $0 \leq c, c(y) \leq 1$ and $y \mapsto c(y)$ is measurable, being a
  supremum over $L$ of infima over the countably many windows of length $\geq L$ of measurable
  functions (Definition~\ref{def:uniform-liminf}).

  \emph{A sequence of windows realizing $c$.}  For each $k \geq 1$ the set of windows
  $[M,N)$ with $N - M \geq k$ and $\unifavg \int c_n\,d\nu \leq c + 1/k$ is non-empty, since
  otherwise $\inf\{\text{averages over windows of length} \geq k\} \geq c + 1/k$, forcing
  $c \geq c + 1/k$.  Pick one, $[M_k,N_k)$.  Also, by definition of the liminf, for each $j$
  there is $L_j$ with $\unifavg \int c_n\,d\nu \geq c - 1/j$ for all windows of length $\geq L_j$;
  since $N_k - M_k \geq k \to \infty$, the averages along $[M_k,N_k)$ converge to $c$.

  \emph{Comparison fibrewise.}  Put $F_k(y) \coloneqq \avg{M_k \leq n < N_k} c_n(y) \in [0,1]$,
  measurable, and $\ell(y) \coloneqq \liminf_k F_k(y)$.  Then $\ell(y) \geq c(y)$: for each
  $j$, since $N_k - M_k \to \infty$, eventually $N_k - M_k \geq L_j(y)$ where $L_j(y)$ is such
  that all averages of $c_\cdot(y)$ over windows of length $\geq L_j(y)$ are $\geq c(y) - 1/j$;
  so $\liminf_k F_k(y) \geq c(y) - 1/j$ for every $j$.

  \emph{Fatou.}  The $F_k$ are non-negative and measurable, so
  $\int \ell\,d\nu = \int \liminf_k F_k\,d\nu \leq \liminf_k \int F_k\,d\nu
   = \liminf_k \avg{M_k\leq n<N_k}\int c_n\,d\nu = c$ (\texttt{lintegral\_liminf\_le}, after
  passing to \texttt{ENNReal}; all functions are bounded by $1$, so the Bochner and Lebesgue
  integrals agree).  Combining, $c \geq \int \ell\,d\nu \geq \int c(y)\,d\nu(y)$.
\end{proof}

\begin{remark}
  Fatou's lemma is the only place where the uniformity of the CSZ property costs anything: a
  uniform liminf is not a liminf along a single sequence, so Fatou cannot be applied to it
  directly.  The device above --- extract one sequence of intervals realizing the uniform
  liminf, apply Fatou along it, and compare with the uniform liminf fibrewise --- is what the
  Lean proof must reproduce.  Note that it needs the uniform liminf to be a liminf along a
  countably generated filter, which \texttt{uniformFilter} is (\texttt{uniformFilter\_hasBasis}
  with the countable index $L \in \NN$).
\end{remark}

\begin{proposition}
  \label{prop:reduction}
  \lean{CantorSzemeredi.MPS.CSZ_of_forall_ergodic_CSZ}
  \uses{thm:ecsz, def:csz, lem:symbolic-model, lem:ergodic-decomposition, lem:fatou-uniform}
  Suppose every ergodic system satisfying the standing assumption is CSZ.  Then every invertible
  measure preserving system is CSZ.
\end{proposition}

\begin{proof}
  \uses{thm:ecsz, def:csz, lem:symbolic-model, lem:ergodic-decomposition, lem:fatou-uniform}
  Fix $(X,\Abb,m,T)$, a Cantor set $\Cantor$ of base $b$, integers $t \geq 1$, $i \geq 0$, and
  $A \in \Abb$ with $m(A) > 0$.  Let $(\Omega,\mathcal B,\mu,\tau)$, $\widetilde A$ be the
  symbolic model of Lemma~\ref{lem:symbolic-model}.  By \eqref{eq:model} with
  $d_s = sb^ik_n$, the multiple correlations of $A$ in $X$ and of $\widetilde A$ in $\Omega$
  coincide for every $n$, so it suffices to prove
  $\unifliminf\unifavg \mu(\bigcap_{s} \tau^{-sb^ik_n}\widetilde A) > 0$; and
  $\mu(\widetilde A) = m(A) > 0$.

  Let $(\mu_y)$ be the ergodic decomposition of Lemma~\ref{lem:ergodic-decomposition}, and
  for a $\tau$-invariant probability $\nu$ write
  $c_n(\nu) \coloneqq \nu(\bigcap_{s=0}^{t}\tau^{-sb^ik_n}\widetilde A) \in [0,1]$.  By (i),
  $c_n(\mu) = \int c_n(\mu_y)\,d\mu(y)$, and $y \mapsto c_n(\mu_y)$ is measurable.

  \emph{The good fibres.}  Since $0 < \mu(\widetilde A) = \int \mu_y(\widetilde A)\,d\mu(y)$, the
  set $Y_+ \coloneqq \{y : \mu_y(\widetilde A) > 0\}$ has $\mu(Y_+) > 0$ (the integral of a
  non-negative function vanishing a.e.\ is $0$).  Discard the null set where (ii) or (iii)
  fails.  For $y \in Y_+$ the system $(\Omega,\mathcal B,\mu_y,\tau)$ is an invertible measure
  preserving system which is ergodic and on a compact metric space with its Borel
  $\sigma$-algebra, so by hypothesis it is CSZ; applied to $\widetilde A$, with the same
  $\Cantor, t, i$,
  \[
    c(y) \coloneqq \unifliminf\unifavg c_n(\mu_y) > 0 , \qquad y \in Y_+ .
  \]

  \emph{Fatou.}  By Lemma~\ref{lem:fatou-uniform} with $c_n(y) = c_n(\mu_y)$,
  \[
    \unifliminf\unifavg c_n(\mu) \geq \int_\Omega c(y)\,d\mu(y) \geq \int_{Y_+} c(y)\,d\mu(y) > 0 ,
  \]
  the last because $c > 0$ on the set $Y_+$ of positive measure
  (\texttt{integral\_pos\_iff\_support\_of\_nonneg\_ae}).  As $\Cantor, t, i, A$ were arbitrary,
  $(X,\Abb,m,T)$ is CSZ.
\end{proof}

\section{The correspondence principle and the main theorems}

\begin{theorem}[Furstenberg correspondence principle]
  \label{thm:furstenberg-correspondence}
  \lean{CantorSzemeredi.furstenberg_correspondence}
  \uses{def:banach-density, def:mps, lem:symbolic-model}
  Let $A \subset \ZZ$ with $\dstar(A) > 0$.  Then there are an invertible measure preserving
  system $(X,\Abb,m,T)$ and $\tilde A \in \Abb$ with $m(\tilde A) = \dstar(A)$ such that for every
  finite collection of integers $r_1,\dots,r_j$,
  \begin{equation}
    \label{eq:correspondence}
    \dstar\bigl( A \cap (A-r_1) \cap \cdots \cap (A-r_j) \bigr)
      \geq m\bigl( \tilde A \cap T^{-r_1}\tilde A \cap \cdots \cap T^{-r_j}\tilde A \bigr) .
  \end{equation}
\end{theorem}

\begin{proof}
  \uses{def:banach-density, lem:symbolic-model}
  Let $d \coloneqq \dstar(A) > 0$.  Work in $\Omega = \{0,1\}^{\ZZ}$ with the shift $\tau$ and
  the clopen set $\widetilde A = \{\omega : \omega(0) = 1\}$ of Lemma~\ref{lem:symbolic-model}(i);
  let $\omega_A \coloneqq \one_A \in \Omega$.  Note $\tau^n\omega_A \in \tau^{-r}\widetilde A$
  iff $\omega_A(n + r) = 1$ iff $n + r \in A$.

  \emph{Windows realizing the density.}  By Definition~\ref{def:banach-density} there are
  windows $[M_k,N_k)$ of integers with $N_k - M_k \to \infty$ and
  $\abs{A \cap [M_k,N_k)}/(N_k - M_k) \to d$ (a limsup along a countably generated filter is
  attained along a sequence).

  \emph{Empirical measures.}  Let
  $\nu_k \coloneqq \avg{M_k \leq n < N_k} \delta_{\tau^n\omega_A}$, a Borel probability measure on
  $\Omega$.  By (i) and Mathlib's Prokhorov theorem in the form
  \texttt{instCompactSpaceProbabilityMeasure} --- the space of Borel probability measures on a
  compact space is compact in the weak topology --- and since $\Omega$ is metrizable, so that
  this space is metrizable as well (\texttt{ProbabilityMeasure} on a metrizable space is
  metrizable via the L\'evy--Prokhorov metric) and compactness is sequential, there is a
  subsequence $\nu_{k_j}$ converging weakly to a probability measure $\mu$:
  $\int \varphi\,d\nu_{k_j} \to \int\varphi\,d\mu$ for every bounded continuous $\varphi$.  In
  particular, for every clopen $B$ (whose indicator is continuous), $\nu_{k_j}(B) \to \mu(B)$.

  \emph{Invariance.}  For bounded continuous $\varphi$,
  \[
    \Bigl\lvert \int \varphi\circ\tau\,d\nu_k - \int\varphi\,d\nu_k \Bigr\rvert
      = \frac{1}{N_k - M_k}\bigl\lvert \varphi(\tau^{N_k}\omega_A) - \varphi(\tau^{M_k}\omega_A) \bigr\rvert
      \leq \frac{2\norm\varphi_\infty}{N_k - M_k} \to 0 ,
  \]
  so $\int\varphi\circ\tau\,d\mu = \int\varphi\,d\mu$ for all bounded continuous $\varphi$, i.e.\
  $\tau_\ast\mu = \mu$ (two Borel probability measures on a metrizable space agreeing on all
  bounded continuous functions are equal: \texttt{ext\_of\_forall\_lintegral\_eq\_of\_IsFiniteMeasure}
  or the characterization of weak convergence).  Thus $(\Omega,\mathcal B,\mu,\tau)$ is an
  invertible measure preserving system; take it as $(X,\Abb,m,T)$ and $\tilde A = \widetilde A$.

  \emph{The measure of $\widetilde A$.}  $\nu_k(\widetilde A) = \abs{A \cap [M_k,N_k)}/(N_k-M_k) \to d$,
  and $\widetilde A$ is clopen, so $\mu(\widetilde A) = d$.

  \emph{The inequality.}  Fix $r_1,\dots,r_j$ and let
  $B \coloneqq \widetilde A \cap \tau^{-r_1}\widetilde A \cap\cdots\cap\tau^{-r_j}\widetilde A$, clopen as
  a finite intersection of clopen sets.  Then
  \[
    \nu_k(B) = \frac{\abs{\{n \in [M_k,N_k) : n \in A,\ n + r_1 \in A, \dots, n + r_j \in A\}}}{N_k - M_k}
      = \frac{\abs{A' \cap [M_k,N_k)}}{N_k - M_k} , \qquad A' \coloneqq A \cap (A - r_1) \cap \cdots \cap (A - r_j) .
  \]
  Since $N_{k_j} - M_{k_j} \to \infty$, $\limsup_j \nu_{k_j}(B) \leq \dstar(A')$ by the definition
  of the upper Banach density as a limsup along \texttt{uniformFilterInt}.  But
  $\nu_{k_j}(B) \to \mu(B)$, so $\mu(B) \leq \dstar(A')$, which is \eqref{eq:correspondence}.
\end{proof}

\begin{remark}[Alternatives for the compactness step]
  The weak-$\ast$ limit can be replaced by an elementary diagonal argument if one prefers to
  avoid the topology on measures: the cylinder sets are countable, so a subsequence can be
  chosen along which $\nu_k(C)$ converges for every cylinder $C$; the limit is finitely
  additive on the clopen algebra, and countably additive there because a countable disjoint
  union of clopen sets which is clopen is finite (compactness); Carath\'eodory extends it to
  $\mathcal B$.  This is longer.  Mathlib's \texttt{ProbabilityMeasure} compactness is the
  recommended route; the only care needed is that the weak topology used there is defined by
  integration against bounded continuous functions, which for clopen $B$ includes $\one_B$.
\end{remark}

\begin{proof}[Proof of Theorem~\ref{thm:main-ergodic}]
  \proves{thm:main-ergodic}
  \uses{thm:ecsz, prop:reduction, lem:csz-functions, def:csz}
  By Proposition~\ref{prop:reduction} and Theorem~\ref{thm:ecsz}, every invertible measure
  preserving system is CSZ.  Fix such a system, $\ell \geq 1$, and a non-negative bounded
  measurable $f$ not a.e.\ zero.  Apply Lemma~\ref{lem:csz-functions} (for the top factor, at
  $t = \ell$, $i = 0$, and the given $\Cantor$): it converts the set form \eqref{eq:csz} into
  \[
    \unifliminf\unifavg \int_X \prod_{j=0}^{\ell} T^{jk_n}f\,dm > 0 ,
  \]
  since $b^0k_n = k_n$.  That is Theorem~\ref{thm:main-ergodic}.
\end{proof}
